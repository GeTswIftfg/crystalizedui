-- print/show a tiny profiler summary: slowest + average callback times
function Crystalized:DebugReport()
	if #self._profile == 0 then
		self:Notify({ Title = "Profiler", Content = "No samples yet - toggle Debug on and click around", Duration = 3 })
		return
	end
	local sum, worst, wsrc = 0, 0, "?"
	for _, s in ipairs(self._profile) do
		sum = sum + s.Ms
		if s.Ms > worst then worst, wsrc = s.Ms, s.Src end
	end
	local avg = sum / #self._profile
	print(string.format("[Crystalized] Profiler: %d samples, avg %.2fms, worst %.1fms (%s)",
		#self._profile, avg, worst, wsrc))
	self:Notify({ Title = "Profiler", Type = worst > 16 and "warn" or "success",
		Content = string.format("%d samples - avg %.1fms - worst %.1fms", #self._profile, avg, worst),
		Duration = 4 })
end

-- icon lookup: accepts a map name, an "rbxassetid://..." url, or nil
local function ResolveIcon(s)
	if type(s) ~= "string" then return nil end
	if Crystalized.Icons[s] then return Crystalized.Icons[s] end
	if s:match("^rbxassetid://%d+$") then return s end
	return nil
end

-- Accepts a Color3, an accent preset name ("Azure"), a named color ("Green"),
-- or a hex string ("#ff2d55") -> Color3 (nil if it can't be resolved)
local function ResolveColor(c)
	if typeof(c) == "Color3" then return c end
	if type(c) == "string" then
		if Crystalized.Accents[c] then return Crystalized.Accents[c] end
		if NAMED_COLORS[c] then return NAMED_COLORS[c] end
		local hex = string.gsub(c, "^#", "")
		if #hex == 6 and hex:match("^[%x]+$") then
			return Color3.fromRGB(
				tonumber(hex:sub(1, 2), 16), tonumber(hex:sub(3, 4), 16), tonumber(hex:sub(5, 6), 16))
		end
	end
	return nil
end

-- Live theme table (mutated in place so all references stay valid)
local Theme = { Name = "Dark", Accent = Crystalized.Accents.Crimson }
Crystalized.Theme = Theme

-- Lightens (or darkens with negative amt) a color -- used for hover states
local function Lighten(color, amt)
	return Color3.new(
		color.R + (amt > 0 and (1 - color.R) * amt or color.R * amt),
		color.G + (amt > 0 and (1 - color.G) * amt or color.G * amt),
		color.B + (amt > 0 and (1 - color.B) * amt or color.B * amt)
	)
end

-- Rebuilds the live Theme table from base theme + accent
local function RebuildTheme()
	local base = BaseThemes[Theme.Name] or BaseThemes.Dark
	for key, value in pairs(base) do Theme[key] = value end
	Theme.AccentSoft = Lighten(Theme.Accent, 0.18)
	Theme.Hover      = Lighten(Theme.Element, Theme.Name ~= "Light" and 0.12 or -0.08)
	Theme.Gold       = Color3.fromRGB(255, 199, 64)   -- premium accents, same in both themes
end
RebuildTheme()

-- Quick Instance factory: Create("Frame", {Size=...}, {child1, child2})
-- Font swap system: every text element registers here, SetFont() restyles all
local FONT_FAMILIES = {
	["Gotham"]     = { Face = Enum.Font.Gotham,     Bold = Enum.Font.GothamSemibold },
	["Arial"]      = { Face = Enum.Font.Arial,      Bold = Enum.Font.ArialBold },
	["Code"]       = { Face = Enum.Font.Code,       Bold = Enum.Font.Code },
	["FredokaOne"] = { Face = Enum.Font.FredokaOne, Bold = Enum.Font.FredokaOne }, -- chunky, Minecraft-ish
	["Bangers"]    = { Face = Enum.Font.Bangers,    Bold = Enum.Font.Bangers },
}
local textRegistry = {}  -- { { i=inst, b=isBold }, ... }
local activeFont = nil   -- becomes a FONT_FAMILIES entry after SetFont()

local function fontWeightOf(inst)
	local name = tostring(inst.Font)
	return (name:find("Bold") ~= nil) or (name:find("Black") ~= nil) or (name:find("Semi") ~= nil)
end

local function registerText(inst)
	local entry = { i = inst, b = fontWeightOf(inst) }
	table.insert(textRegistry, entry)
	if activeFont then
		entry.i.Font = (entry.b and activeFont.Bold) or activeFont.Face
	end
end

function Crystalized:SetFont(name)
	local fam = FONT_FAMILIES[tostring(name)] or FONT_FAMILIES.Gotham
	activeFont = fam
	for _, t in ipairs(textRegistry) do
		if t.i and t.i.Parent ~= nil then
			t.i.Font = (t.b and fam.Bold) or fam.Face
		end
	end
end

local function Create(className, props, children)
	local inst = Instance.new(className)
	for key, value in pairs(props or {}) do
		if key ~= "Parent" then
			pcall(function() inst[key] = value end)
		end
	end
	for _, child in ipairs(children or {}) do
		child.Parent = inst
	end
	if className == "TextLabel" or className == "TextButton" or className == "TextBox" then
		registerText(inst)
	end
	if props and props.Parent then inst.Parent = props.Parent end
	return inst
end

-- The house motion style: crisp Quint at slightly-faster-than-spec speeds.
-- style = "Snappy" (same Quint, tighter), style = "Pop" reserved for the
-- signature punchy moments (window open, pill pop, premium card).
local function Tween(obj, duration, props, style)
	local easing = (style == "Pop")
		and TweenInfo.new(duration, Enum.EasingStyle.Back,  Enum.EasingDirection.Out)
		or  TweenInfo.new(duration * 0.75, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
	local tw = TweenService:Create(obj, easing, props)
	tw:Play()
	return tw
end

-- Applies a theme color to a registered entry. Called by Paint() at
-- creation and by ApplyTheme() on every live recolor.
local function Recolor(entry, instant)
	local color = entry.c or Theme[entry.r]  -- entry.c = pinned element color (Danger/Color overrides)
	if not color then return end
	local alpha = entry.a
	if alpha == nil then
		alpha = (RoleAlpha[Theme.Name] and RoleAlpha[Theme.Name][entry.r]) or 0
	end
	-- global UI Transparency slider stacks extra see-through on every panel
	if entry.p == "BackgroundColor3" then
		alpha = math.min(0.97, alpha + (Crystalized._extraBG or 0))
	end
	pcall(function()
		if instant then
			entry.i[entry.p] = color
			if entry.p == "BackgroundColor3" then entry.i.BackgroundTransparency = alpha end
			if entry.p == "ImageColor3" then entry.i.ImageTransparency = alpha end
		else
			local props = { [entry.p] = color }
			if entry.p == "BackgroundColor3" then props.BackgroundTransparency = alpha end
			if entry.p == "ImageColor3" then props.ImageTransparency = alpha end
			TweenService:Create(entry.i, TweenInfo.new(0.25, Enum.EasingStyle.Quint), props):Play()
		end
	end)
end

-- Paints an instance property with a theme color AND remembers it so
-- ApplyTheme() can live-recolor everything later. Returns the entry so
-- elements can mutate entry.r / entry.a and call Recolor() themselves.
local function Paint(inst, prop, role, alpha)
	local entry = { i = inst, p = prop, r = role, a = alpha }
	Recolor(entry, true)
	table.insert(Crystalized._paints, entry)
	return entry
end

-- Soft drop-shadow follower for floating cards/popups (per answer 25).
-- Tracks the target's size/position every frame, cleans itself up.
local function AddShadow(target, pad, transparency)
	pad = pad or 30
	if not target or not target.Parent then return end
	local shadow = Create("ImageLabel", {
		AnchorPoint = target.AnchorPoint,
		Position = UDim2.new(target.Position.X.Scale, target.Position.X.Offset,
			target.Position.Y.Scale, target.Position.Y.Offset + 3),
		Size = UDim2.new(0, target.AbsoluteSize.X + pad * 2, 0, target.AbsoluteSize.Y + pad * 2),
		BackgroundTransparency = 1, ZIndex = math.max(0, target.ZIndex - 1),
		Image = "rbxassetid://5028857084", ScaleType = Enum.ScaleType.Slice,
		SliceCenter = Rect.new(24, 24, 276, 276), ImageColor3 = Color3.new(0, 0, 0),
		ImageTransparency = transparency or 0.6, Parent = target.Parent,
	})
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not shadow.Parent or not target.Parent then
			pcall(function() conn:Disconnect() end)
			if shadow.Parent then shadow:Destroy() end
			return
		end
		shadow.Visible = target.Visible
		shadow.Position = UDim2.new(target.Position.X.Scale, target.Position.X.Offset,
			target.Position.Y.Scale, target.Position.Y.Offset + 3)
		shadow.Size = UDim2.new(0, target.AbsoluteSize.X + pad * 2, 0, target.AbsoluteSize.Y + pad * 2)
	end)
	return shadow
end

-- Extra see-through for every panel (0..60 percent). Stacks on RoleAlpha;
-- Recolor() reads Crystalized._extraBG, this just applies it live.
function Crystalized:SetTransparency(pct)
	self._extraBG = math.clamp((tonumber(pct) or 0) / 100, 0, 0.6)
	for _, entry in ipairs(self._paints) do
		if entry.i and entry.i.Parent ~= nil and entry.p == "BackgroundColor3" then
			Recolor(entry, true)
		end
	end
end

-- Background artwork: accepts "rbxassetid://123", "123" or an asset url.
-- Empty string removes it. The holder is created with the window (BGHolder).
function Crystalized:SetBackgroundImage(id)
	local digits = tostring(id or ""):gsub("[^%d]", "")
	local holder = self._bgHolder
	if not holder then self._pendingBG = digits ~= "" and digits or nil return end
	if digits == "" then
		local old = holder:FindFirstChildWhichIsA("ImageLabel")
		if old then old:Destroy() end
		return
	end
	local img = holder:FindFirstChildWhichIsA("ImageLabel")
	if not img then
		img = Create("ImageLabel", {
			Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1,
			ScaleType = Enum.ScaleType.Crop, ImageTransparency = 0.85, ZIndex = 1,
			Parent = holder,
		})
	end
	img.Image = "rbxassetid://" .. digits
end

Crystalized._themeHooks = {}
-- Recolors the ENTIRE ui (theme swap / accent swap)
local function ApplyTheme()
	for _, entry in ipairs(Crystalized._paints) do
		if entry.i and entry.i.Parent ~= nil then
			Recolor(entry, false)
		end
	end
	-- Ask every open window to refresh its custom states (tabs, switches)
	for _, fn in ipairs(Crystalized._themeHooks) do
		task.spawn(fn)
	end
end

-- Rounds to n decimals: Round(3.14159, 2) -> 3.14
local function Round(num, decimals)
	local mult = 10 ^ (decimals or 0)
	return math.floor(num * mult + 0.5) / mult
end

-- Fires a user callback SAFELY (errors never break the ui)
local function Fire(fn, ...)
	if type(fn) ~= "function" then return end
	local args = { ... }
	local label = debug and debug.getinfo(fn, "S") and debug.getinfo(fn, "S").short_src or "?"
	task.spawn(function()
		local t0 = os.clock()
		local ok, err = pcall(fn, table.unpack(args))
		local ms = (os.clock() - t0) * 1000
		-- lightweight callback profiler (per answer 24): enable with
		-- Crystalized.Debug = true -> slow callbacks get flagged in the console,
		-- and Crystalized:DebugReport() prints averages.
		if Crystalized.Debug then
			local p = Crystalized._profile
			p[#p + 1] = { Src = label, Ms = ms }
			if #p > 500 then table.remove(p, 1) end
			if ms > 16 then
				warn(string.format("[Crystalized] Slow callback: %s took %.1fms", label, ms))
			end
		end
		if not ok then warn("[Crystalized] Callback error: " .. tostring(err)) end
	end)
end

-- Makes `frame` draggable by dragging `handle` (mouse + touch)
local function MakeDraggable(handle, frame)
	if not handle or not frame then return end
	local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
	local totalDrag = 0
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
			dragging  = true
			dragStart = input.Position
			startPos  = frame.Position
			totalDrag = 0
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					-- Click on Pill opens UI if moved < 6px
					if frame == Pill and totalDrag < 6 then
						if Crystalized._currentWindow and Crystalized._currentWindow.SetVisible then
							if not Crystalized._currentWindow._visible then
								Crystalized._currentWindow:SetVisible(true)
							end
						end
						return
					end
					local cam = workspace.CurrentCamera
					if cam and frame:GetAttribute("IsMainWindow") then
						local vp = cam.ViewportSize
						local abs, size = frame.AbsolutePosition, frame.AbsoluteSize
						-- swipe-to-minimize: fling/slide the UI off screen and it fades away
						-- in that direction and minimizes to the pill (Android tab style)
						local offLeft   = (abs.X + size.X * 0.45 < 0)
						local offRight  = (abs.X + size.X * 0.55 > vp.X)
						local offTop    = (abs.Y + size.Y * 0.45 < 0)
						local offBottom = (abs.Y + size.Y * 0.55 > vp.Y)

						if offLeft or offRight or offTop or offBottom then
							local targetX = abs.X + (offLeft and -260 or offRight and 260 or 0)
							local targetY = abs.Y + (offTop and -200 or offBottom and 200 or 0)
							local flyTween = TweenService:Create(frame, TweenInfo.new(0.24, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
								Position = UDim2.new(0, targetX + size.X * frame.AnchorPoint.X, 0, targetY + size.Y * frame.AnchorPoint.Y),
							})
							flyTween:Play()
							Fade(frame, true, 0.22)
							task.delay(0.25, function()
								frame.Position = UDim2.new(0.5, 0, 0.5, 0)
								if Crystalized._currentWindow and Crystalized._currentWindow.SetVisible then
									Crystalized._currentWindow:SetVisible(false)
								end
							end)
							return
						end

						-- snap-to-edge docking (per answer 21): near an edge and releasing
						-- glides the window perfectly flush to it
						local nx, ny = abs.X, abs.Y
						local snapped = false
						if abs.X < 24 then nx = 12 ; snapped = true
						elseif abs.X + size.X > vp.X - 24 then nx = vp.X - size.X - 12 ; snapped = true end
						if abs.Y < 24 then ny = 12 ; snapped = true
						elseif abs.Y + size.Y > vp.Y - 24 then ny = vp.Y - size.Y - 12 ; snapped = true end
						if snapped then
							TweenService:Create(frame, TweenInfo.new(0.22, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
								Position = UDim2.new(0, nx + size.X * frame.AnchorPoint.X, 0, ny + size.Y * frame.AnchorPoint.Y),
							}):Play()
						end
					end
				end
			end)
		end
	end)
	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging and dragStart and startPos then
			local delta = input.Position - dragStart
			totalDrag = math.max(totalDrag, delta.Magnitude)
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

--============================================================--
-- FADE SYSTEM (windows fade out instead of snapping shut)
-- Remembers every instance's original transparency once, so we
-- can always restore cosmetics exactly.
--============================================================--
local FadeProps = {
	Frame          = { "BackgroundTransparency" },
	ScrollingFrame = { "BackgroundTransparency", "ScrollBarImageTransparency" },
	TextLabel      = { "BackgroundTransparency", "TextTransparency" },
	TextButton     = { "BackgroundTransparency", "TextTransparency" },
	TextBox        = { "BackgroundTransparency", "TextTransparency" },
	ImageLabel     = { "BackgroundTransparency", "ImageTransparency" },
	ImageButton    = { "BackgroundTransparency", "ImageTransparency" },
	UIStroke       = { "Transparency" },
}

local function Fade(root, fadeOut, duration)
	-- bump a sequence counter so a stale "hide" never kills a re-shown ui
	Crystalized._fadeSeq[root] = (Crystalized._fadeSeq[root] or 0) + 1
	local mySeq = Crystalized._fadeSeq[root]
	local instances = { root }
	for _, d in ipairs(root:GetDescendants()) do table.insert(instances, d) end
	for _, inst in ipairs(instances) do
		local props = FadeProps[inst.ClassName]
		if props then
			if not Crystalized._fades[inst] then
				Crystalized._fades[inst] = {}
				for _, p in ipairs(props) do
					pcall(function() Crystalized._fades[inst][p] = inst[p] end)
				end
			end
			for _, p in ipairs(props) do
				local target = fadeOut and 1 or (Crystalized._fades[inst][p] or 0)
				TweenService:Create(inst, TweenInfo.new(duration, Enum.EasingStyle.Quint), { [p] = target }):Play()
			end
		end
	end
	if fadeOut then
		task.delay(duration + 0.05, function()
			if Crystalized._fadeSeq[root] == mySeq then root.Visible = false end
		end)
	end
end

local tooltipToken = 0
local function AttachTooltip(row, text)
	if not text or text == "" then return end
	-- optional affordance: a dim "?" affordance at the end of the row's name
	-- so players can see at a glance that a description exists
	if row:GetAttribute("IsRow") and not row:GetAttribute("HelpDot") then
		row:SetAttribute("HelpDot", true)
		for _, ch in ipairs(row:GetChildren()) do
			if ch:IsA("TextLabel") and ch.BackgroundTransparency == 1 and ch.TextSize >= 12 then
				ch.RichText = true
				ch.Text = ch.Text .. '  <font color="rgb(130,130,145)">?</font>'
				break
			end
		end
	end
	row.MouseEnter:Connect(function()
		tooltipToken = tooltipToken + 1
		local my = tooltipToken
		task.delay(0.35, function()
			if my ~= tooltipToken then return end
			TooltipLabel.Text = text
			Tooltip.Visible = true
			Tooltip.BackgroundTransparency = 1
			TooltipLabel.TextTransparency = 1
			Tween(Tooltip, 0.2, { BackgroundTransparency = 0.02 }, "Snappy")
			Tween(TooltipLabel, 0.2, { TextTransparency = 0 }, "Snappy")
		end)
	end)
	row.MouseLeave:Connect(function()
		tooltipToken = tooltipToken + 1
		Tooltip.Visible = false
	end)
end

local function NotifyNow(opts)
	opts = opts or {}
	local duration = opts.Duration or 3.5
	local wrapper = Create("Frame", {

			PaddingTop = UDim.new(0, 8), PaddingBottom = UDim.new(0, 8),
			PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 10),
		}),

	Paint(stroke, "Color", "Stroke")

	local typeInfo = {
		success = { c = Color3.fromRGB(107, 229, 133) },
		info    = { c = Color3.fromRGB(79, 163, 255) },
		warn    = { c = Color3.fromRGB(255, 199, 64) },
		error   = { c = Color3.fromRGB(255, 71, 87) },
	}
	local tinfo = opts.Type and typeInfo[string.lower(tostring(opts.Type))]
	if not tinfo then
		local custom = ResolveColor(opts.Color)
		tinfo = custom and { c = custom } or nil
	end

	-- slim accent bar on the left
	local bar = Create("Frame", {
		Size = UDim2.new(0, 2.5, 1, -10), Position = UDim2.new(0, 0, 0, 5),
		BorderSizePixel = 0, Parent = card,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	if tinfo then bar.BackgroundColor3 = tinfo.c
	else Paint(bar, "BackgroundColor3", "Accent") end

	-- slim progress bar along the bottom of the toast
	local ptrack = Create("Frame", {
		AnchorPoint = Vector2.new(0, 1), Position = UDim2.new(0, 6, 1, -2),
		Size = UDim2.new(1, -12, 0, 2), BackgroundTransparency = 0.8,
		BorderSizePixel = 0, Parent = wrapper, ZIndex = 7,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	ptrack.BackgroundColor3 = Color3.new(0, 0, 0)
	local pfill = Create("Frame", {
		Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 0,
		BorderSizePixel = 0, Parent = ptrack, ZIndex = 8,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	if tinfo then
		pfill.BackgroundColor3 = tinfo.c
	else
		Paint(pfill, "BackgroundColor3", "Accent")
	end
	local title = Create("TextLabel", {
		Size = UDim2.new(1, 0, 0, 14), BackgroundTransparency = 1,
		Font = Enum.Font.GothamBold, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left,
		Text = tostring(opts.Title or "Notification"), LayoutOrder = 1, Parent = card,
	})
	if tinfo then title.TextColor3 = tinfo.c
	else Paint(title, "TextColor3", "Text") end
	if opts.Content and opts.Content ~= "" then
		local body = Create("TextLabel", {
			Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,

			Text = tostring(opts.Content), LayoutOrder = 2, Parent = card,

	end

	Tween(card, 0.5, { Position = UDim2.new(0, 0, 0, 0) })

	TweenService:Create(pfill, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
		Size = UDim2.new(0, 0, 1, 0),

		out.Completed:Wait()
		wrapper:Destroy()
		notifActive = notifActive - 1
		if #notifQueue > 0 then
			local nextUp = table.remove(notifQueue, 1)
			Crystalized:Notify(nextUp)
		end
	end)
end

-- Crystalized:Notify({ Title, Content, Type, Duration })
function Crystalized:Notify(opts)
	opts = opts or {}
	if notifActive >= NOTIF_CAP then -- stack is full -> wait your turn
		table.insert(notifQueue, opts)
		return
	end
	notifActive = notifActive + 1
	NotifyNow(opts)
end

--============================================================--
-- KEYBIND CAPTURE (click a bind chip, press a key)
--============================================================--
local capturing = nil   -- the bind entry currently waiting for a key

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if capturing then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			if input.KeyCode == Enum.KeyCode.Escape then
				-- cancel capture, keep old key
			elseif input.KeyCode == Enum.KeyCode.Backspace then
				capturing.Key = nil
				capturing.Update()
			else
				capturing.Key = input.KeyCode
				capturing.Update()
			end
			capturing = nil
			RefreshBindPanel()
		end
		return
	end
	if gameProcessed then return end
	if input.UserInputType == Enum.UserInputType.Keyboard then
		for _, bind in ipairs(Crystalized._binds) do
			if bind.Key == input.KeyCode and not bind._held then
				bind._held = true
				if bind.Mode == "Hold" then
					-- HOLD mode: key down = active, key up = inactive
					if bind.OnHoldStart then Fire(bind.OnHoldStart) else Fire(bind.OnPress) end
				else
					Fire(bind.OnPress)
				end
			end
		end
	end
end)

-- HOLD-mode binds deactivate on key release
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Keyboard then
		for _, bind in ipairs(Crystalized._binds) do
			if bind._held and bind.Key == input.KeyCode then
				bind._held = false
				if bind.Mode == "Hold" and bind.OnHoldEnd then Fire(bind.OnHoldEnd) end
			end
		end
	end
end)

-- Creates one bind entry. getState is optional (drives the accent dot).
-- Mode = "Press" (tap = toggle, default) or "Hold" (active only while held).
local function RegisterBind(name, key, onPress, getState)
	local bind = { Name = name, Key = key, Mode = "Press", _held = false,
		OnPress = onPress, GetState = getState, Update = function() end, Refresh = function() end }
	table.insert(Crystalized._binds, bind)
	RefreshBindPanel()
	return bind
end

-- switches a bind between Press and Hold (saves it to configs too)
local function SetBindMode(bind, mode)
	if not bind then return end
	if mode ~= "Hold" and mode ~= "Press" then return end
	bind.Mode = mode
	if bind.ModeFlag then Crystalized.Flags[bind.ModeFlag] = mode end
	bind.Refresh()
	RefreshBindPanel()
end

--============================================================--
-- CONFIG SYSTEM (Obsidian-style; uses executor file functions
-- with a graceful in-memory fallback)
--============================================================--
local CONFIG_FOLDER = "CrystalizedConfigs"
local memoryConfigs = {}   -- fallback store if file funcs are missing

local function FileSafe(fn, ...)
	if type(fn) ~= "function" then return nil end
	local ok, res = pcall(fn, ...)
	if ok then return res end
	return nil
end

local function EnsureFolder()
	if type(isfolder) == "function" and type(makefolder) == "function" then
		if not FileSafe(isfolder, CONFIG_FOLDER) then FileSafe(makefolder, CONFIG_FOLDER) end
	end
end

-- Serializes/deserializes values that JSON can't hold natively
local function Serialize(value)
	if typeof(value) == "Color3" then
		return { __t = "Color3", math.floor(value.R * 255), math.floor(value.G * 255), math.floor(value.B * 255) }
	elseif typeof(value) == "EnumItem" then
		return { __t = "EnumItem", tostring(value.EnumType), value.Name }
	end
	return value
end

local function Deserialize(value)
	if type(value) == "table" and value.__t == "Color3" then
		return Color3.fromRGB(value[1], value[2], value[3])
	elseif type(value) == "table" and value.__t == "EnumItem" then
		local ok, item = pcall(function() return Enum[value[1]][value[2]] end)
		if ok then return item end
		return nil
	end
	return value
end

-- Crystalized:SaveConfig("main")
function Crystalized:SaveConfig(name)
	name = tostring(name or "main")
	local data = {
		Theme  = Theme.Name,
		Accent = { Serialize(Theme.Accent) },
		Flags  = {},
	}
	for flag, value in pairs(self.Flags) do
		if type(value) == "table" then
			local arr = {}
			for i, v in ipairs(value) do arr[i] = Serialize(v) end
			data.Flags[flag] = { __t = "List", arr }
		else
			data.Flags[flag] = Serialize(value)
		end
	end
	local json = HttpService:JSONEncode(data)
	EnsureFolder()
	if type(writefile) == "function" then
		FileSafe(writefile, CONFIG_FOLDER .. "/" .. name .. ".json", json)
	else
		memoryConfigs[name] = json
	end
	self:Notify({ Title = "Config", Content = "Saved config \"" .. name .. "\"", Duration = 2 })
end

function Crystalized:LoadConfig(name)
	name = tostring(name or "main")
	local json
	if type(readfile) == "function" and type(isfile) == "function" then
		if FileSafe(isfile, CONFIG_FOLDER .. "/" .. name .. ".json") then
			json = FileSafe(readfile, CONFIG_FOLDER .. "/" .. name .. ".json")
		end
	else
		json = memoryConfigs[name]
	end
	if not json then
		self:Notify({ Title = "Config", Content = "Config \"" .. name .. "\" not found", Duration = 2 })
		return
	end
	local ok, data = pcall(HttpService.JSONDecode, HttpService, json)
	if not ok or type(data) ~= "table" then return end
	self._silent = true
	if data.Theme then self:SetTheme(data.Theme) end
	if data.Accent then
		local accent = Deserialize(data.Accent[1])
		if accent then self:SetAccent(accent) end
	end
	for flag, stored in pairs(data.Flags or {}) do
		local meta = self._flagMeta[flag]
		if meta then
			local value
			if type(stored) == "table" and stored.__t == "List" then
				value = {}
				for i, v in ipairs(stored[1] or {}) do value[i] = Deserialize(v) end
			else
				value = Deserialize(stored)
			end
			if value ~= nil then pcall(meta.Set, value) end
		end
	end
	self._silent = false
	RefreshBindPanel()
	self:Notify({ Title = "Config", Content = "Loaded config \"" .. name .. "\"", Duration = 2 })
end

-- shareable config codes (per answer 19): export -> clipboard code, import -> paste.
-- base64, hand-rolled (no external deps, Luraph-safe)
local B64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local function B64Encode(data)
	local out, i = {}, 1
	while i <= #data do
		local b1 = string.byte(data, i)
		local b2 = i + 1 <= #data and string.byte(data, i + 1) or 0
		local b3 = i + 2 <= #data and string.byte(data, i + 2) or 0
		local n = b1 * 65536 + b2 * 256 + b3
		local c1 = math.floor(n / 262144) % 64
		local c2 = math.floor(n / 4096) % 64
		local c3 = math.floor(n / 64) % 64
		local c4 = n % 64
		out[#out + 1] = string.sub(B64_CHARS, c1 + 1, c1 + 1)
		out[#out + 1] = string.sub(B64_CHARS, c2 + 1, c2 + 1)
		out[#out + 1] = (i + 1 <= #data) and string.sub(B64_CHARS, c3 + 1, c3 + 1) or "="
		out[#out + 1] = (i + 2 <= #data) and string.sub(B64_CHARS, c4 + 1, c4 + 1) or "="
		i = i + 3
	end
	return table.concat(out)
end
local B64_LOOKUP = {}
for i = 1, 64 do B64_LOOKUP[string.sub(B64_CHARS, i, i)] = i - 1 end
local function B64Decode(data)
	data = string.gsub(data, "[^A-Za-z0-9%+%/=]", "")
	local out, acc, bits = {}, 0, 0
	for i = 1, #data do
		local ch = string.sub(data, i, i)
		if ch ~= "=" then
			acc = acc * 64 + (B64_LOOKUP[ch] or 0)
			bits = bits + 6
			if bits >= 8 then
				bits = bits - 8
				out[#out + 1] = string.char(math.floor(acc / 2 ^ bits) % 256)
			end
		end
	end
	return table.concat(out)
end

function Crystalized:ExportConfigCode(name)
	name = tostring(name or "main")
	local json
	if type(readfile) == "function" and type(isfile) == "function"
	and FileSafe(isfile, CONFIG_FOLDER .. "/" .. name .. ".json") then
		json = FileSafe(readfile, CONFIG_FOLDER .. "/" .. name .. ".json")
	else
		json = memoryConfigs[name]
	end
	if not json then
		self:Notify({ Title = "Config", Type = "warn", Content = "Save \"" .. name .. "\" first", Duration = 3 })
		return nil
	end
	local code = "CRYS1-" .. B64Encode(json)
	if type(setclipboard) == "function" then
		pcall(setclipboard, code)
		self:Notify({ Title = "Config", Type = "success", Content = "Code for \"" .. name .. "\" copied - paste it to anyone", Duration = 4 })
	end
	return code
end

function Crystalized:ImportConfigCode(code, saveAs)
	code = tostring(code or "")
	code = string.gsub(code, "^%s*CRYS1%-", "")
	local ok, json = pcall(B64Decode, code)
	if not ok then
		self:Notify({ Title = "Config", Type = "error", Content = "That code is not a Crystalized config", Duration = 3 })
		return false
	end
	local ok2, data = pcall(HttpService.JSONDecode, HttpService, json)
	if not ok2 or type(data) ~= "table" or type(data.Flags) ~= "table" then
		self:Notify({ Title = "Config", Type = "error", Content = "That code is not a Crystalized config", Duration = 3 })
		return false
	end
	saveAs = tostring(saveAs or "imported")
	-- hand it to the same path LoadConfig uses
	-- hand it to the same path LoadConfig uses
	memoryConfigs[saveAs] = json
	if type(writefile) == "function" then
		FileSafe(writefile, CONFIG_FOLDER .. "/" .. saveAs .. ".json", json)
	end
	self:LoadConfig(saveAs)
	return true
end

function Crystalized:DeleteConfig(name)
	name = tostring(name or "main")
	if type(delfile) == "function" and type(isfile) == "function" then
		if FileSafe(isfile, CONFIG_FOLDER .. "/" .. name .. ".json") then
			FileSafe(delfile, CONFIG_FOLDER .. "/" .. name .. ".json")
		end
	else
		memoryConfigs[name] = nil
	end
end

function Crystalized:ListConfigs()
	local list = {}
	if type(listfiles) == "function" then
		EnsureFolder()
		for _, path in ipairs(FileSafe(listfiles, CONFIG_FOLDER) or {}) do
			local name = path:match("([^/\\]+)%.json$")
			if name then table.insert(list, name) end
		end
	else
		for name in pairs(memoryConfigs) do
			if name ~= "__autoload" then table.insert(list, name) end
		end
	end
	table.sort(list)
	return list
end

function Crystalized:SetAutoload(name)
	EnsureFolder()
	if type(writefile) == "function" then
		FileSafe(writefile, CONFIG_FOLDER .. "/autoload.txt", name)
	else
		memoryConfigs.__autoload = name
	end
end

function Crystalized:LoadAutoload()
	local name
	if type(readfile) == "function" and type(isfile) == "function" then
		if FileSafe(isfile, CONFIG_FOLDER .. "/autoload.txt") then
			name = FileSafe(readfile, CONFIG_FOLDER .. "/autoload.txt")
		end
	else
		name = memoryConfigs.__autoload
	end
	if name and name ~= "" then self:LoadConfig(name) end
end

		info.Visible = true
		info.BackgroundTransparency = 1
		info.TextTransparency = 1
		Tween(info, 0.3, { BackgroundTransparency = 0, TextTransparency = 0 }, "Snappy")
	end

	buyBtn.MouseButton1Click:Connect(function()
		if type(setclipboard) == "function" then
			pcall(setclipboard, Crystalized.BuyLink)
			showInfo("<font color=\"#7ce29b\">Copied!</font> <b>" .. Crystalized.BuyLink ..
				"</b> is in your clipboard - paste into your browser to buy.")
		else
			showInfo("Go to: <b>" .. Crystalized.BuyLink .. "</b>")
		end
	end)

	local function shake()
		local baseX = card.Position.X
		for i = 1, 3 do
			TweenService:Create(card, TweenInfo.new(0.05), { Position = UDim2.new(baseX.Scale, baseX.Offset - 7, 0.5, 0) }):Play()
			task.wait(0.05)
			TweenService:Create(card, TweenInfo.new(0.05), { Position = UDim2.new(baseX.Scale, baseX.Offset + 7, 0.5, 0) }):Play()
			task.wait(0.05)
		end
		TweenService:Create(card, TweenInfo.new(0.08), { Position = UDim2.new(baseX.Scale, baseX.Offset, 0.5, card.Position.Y.Offset) }):Play()
	end

	local function tryRedeem()
		if Crystalized:IsKeyValid(keybox.Text) then
			local usedKey = keybox.Text
			dismiss()
			Crystalized:UnlockPremium(usedKey)
		else
			task.spawn(shake)
			keybox.Text = ""
			keybox.PlaceholderText = "Invalid key!"
		end
	end
	redeemBtn.MouseButton1Click:Connect(tryRedeem)
	keybox.FocusLost:Connect(function(enter) if enter then tryRedeem() end end)

	task.defer(function()
		Tween(card.UIScale, 0.45, { Scale = 1 }, "Pop")
	end)
end

-- Makes the base row inside a section + registers search text
local function MakeRow(sectionHolder, height, searchText)
	local row = Create("Frame", {
		Size = UDim2.new(1, 0, 0, height), BorderSizePixel = 0, Parent = sectionHolder,
	}, { Create("UICorner", { CornerRadius = UDim.new(0, 6) }) })
	Paint(row, "BackgroundColor3", "Element")
	row:SetAttribute("IsRow", true)
	if searchText then row:SetAttribute("SearchText", string.lower(searchText)) end
	return row
end

local function MakeNameLabel(parent, name)
	local label = Create("TextLabel", {
		Position = UDim2.new(0, 12, 0, 0), Size = UDim2.new(1, -90, 1, 0),
		BackgroundTransparency = 1, Font = Enum.Font.GothamMedium, TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left, Text = tostring(name), Parent = parent,
	})
	local entry = Paint(label, "TextColor3", "Text")
	return label, entry
end

-- Registers an element's flag so configs can save/restore it
local function RegisterFlag(flag, value, setFn, typeName)
	if not flag then return end
	Crystalized.Flags[flag] = value
	Crystalized._flagMeta[flag] = { Type = typeName, Set = setFn }
end

--> PREMIUM GATE -- call at the end of any builder to gold-lock a row.
-- The row turns gold, gains a PRO badge, and all clicks open the key
-- prompt instead of working, until Crystalized:UnlockPremium() runs.
local function Premiumize(row, nameLabel, nameEntry, featureName)
	if Crystalized.IsPremium then return end
	nameEntry.r = "Gold"
	Recolor(nameEntry, true)
	-- gold PRO badge placed right after the name text
	local w = 90
	pcall(function()
		w = game:GetService("TextService"):GetTextSize(featureName, 13,
			Enum.Font.GothamMedium, Vector2.new(400, 20)).X
	end)
	local badge = Create("TextLabel", {
		AutomaticSize = Enum.AutomaticSize.X, Position = UDim2.new(0, w + 6, 0.5, -7),
		Size = UDim2.new(0, 0, 0, 14), BackgroundColor3 = Theme.Gold,
		TextColor3 = Color3.fromRGB(46, 33, 6), Font = Enum.Font.GothamBold, TextSize = 9,
		Text = "PRO", BorderSizePixel = 0, ZIndex = 3, Parent = nameLabel,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(0, 4) }),
		Create("UIPadding", { PaddingLeft = UDim.new(0, 4), PaddingRight = UDim.new(0, 4) }),
	})
	local glow = Create("UIStroke", { Thickness = 1, Color = Theme.Gold, Transparency = 0.5, Parent = row })
	-- animated gold shine sweep across premium elements
	local pShine = Create("UIGradient", {
		Color = ColorSequence.new({
			ColorSequenceKeypoint.new(0,   Color3.fromRGB(255, 215, 100)),
			ColorSequenceKeypoint.new(0.4, Color3.fromRGB(255, 240, 180)),
			ColorSequenceKeypoint.new(0.6, Color3.fromRGB(255, 199, 64)),
			ColorSequenceKeypoint.new(1,   Color3.fromRGB(255, 215, 100)),
		}),
		Offset = Vector2.new(-1, 0),
		Parent = row,
	})
	task.spawn(function()
		while row and row.Parent and pShine and pShine.Parent do
			pShine.Offset = Vector2.new(-1, 0)
			local t = TweenService:Create(pShine, TweenInfo.new(1.8, Enum.EasingStyle.Linear), {
				Offset = Vector2.new(1, 0) })
			t:Play()
			t.Completed:Wait()
			task.wait(2.8)
		end
	end)
	local blocker = Create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = "",
		ZIndex = 20, AutoButtonColor = false, Parent = row,
	})
	blocker.MouseButton1Click:Connect(function() ShowPremiumPrompt(featureName) end)
	-- undoing the gold when premium unlocks
	table.insert(Crystalized._premiumEls, function()
		badge:Destroy()
		glow:Destroy()
		pcall(function() pShine:Destroy() end)
		blocker:Destroy()
		nameEntry.r = "Text"
		Recolor(nameEntry, false)
	end)
end

--> TOGGLE (with optional keybind chip on the right) ------------
local function BuildToggle(section, opts)
	local row = MakeRow(section._holder, 38, opts.Name)
	local nameLbl, nameEntry = MakeNameLabel(row, opts.Name)
	AttachTooltip(row, opts.Description)
	-- switch
	local switch = Create("Frame", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.new(0, 38, 0, 20), BorderSizePixel = 0, Parent = row,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	local switchEntry = Paint(switch, "BackgroundColor3", "ToggleOff")
	-- per-element styling: Color = "Azure"/"#hex"/Color3, or Danger = true (hot red + !)
	local elemColor = opts.Danger and DANGER_COLOR or ResolveColor(opts.Color)
	if elemColor then switchEntry.c = elemColor end
	if opts.Danger then
		nameLbl.Text = "^  " .. nameLbl.Text
		nameEntry.c = DANGER_COLOR
	end
	local knob = Create("Frame", {
		AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 3, 0.5, 0),
		Size = UDim2.new(0, 14, 0, 14), BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0, Parent = switch,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	local click = Create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = "",
		AutoButtonColor = false, ZIndex = 2, Parent = row,
	})

	local state = opts.Default == true
	local Toggle = {}

	local function apply(animate, fireCallback)
		switchEntry.c = state and elemColor or nil -- custom color only while ON
		switchEntry.r = (state and not elemColor) and "Accent" or "ToggleOff"
		Recolor(switchEntry, not animate)
		local target = state and UDim2.new(1, -17, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
		if animate then Tween(knob, 0.4, { Position = target }) else knob.Position = target end
		if opts.Flag then Crystalized.Flags[opts.Flag] = state end
		if fireCallback then
			if opts.Notify ~= false and not Crystalized._silent then
				Crystalized:Notify({
					Title = tostring(opts.Name),
					Content = state and "Enabled" or "Disabled",
					Duration = 1.6,
				})
			end
			Fire(opts.Callback, state)
		end
		RefreshBindPanel()
	end

	click.MouseButton1Click:Connect(function()
		state = not state
		apply(true, true)
	end)

	function Toggle:Set(value, noCallback)
		state = value == true
		apply(true, not noCallback)
	end
	function Toggle:Get() return state end

	-- optional keybind chip: "keybinds attachable to toggles"
	-- (skipped on touch-only devices -- no keyboard to bind)
	if opts.Keybind and HAS_KEYBOARD then
		local chip = Create("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5),
			Position = UDim2.new(1, -58, 0.5, 0), AutomaticSize = Enum.AutomaticSize.X,
			Size = UDim2.new(0, 0, 0, 20), Text = "", AutoButtonColor = false,
			BorderSizePixel = 0, ZIndex = 3, Parent = row,
		}, {
			Create("UICorner", { CornerRadius = UDim.new(0, 5) }),
			Create("UIPadding", { PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6) }),
		})
		Paint(chip, "BackgroundColor3", "Card")
		local chipLabel = Create("TextLabel", {
			AutomaticSize = Enum.AutomaticSize.X, Size = UDim2.new(0, 0, 1, 0),
			BackgroundTransparency = 1, Font = Enum.Font.GothamSemibold, TextSize = 10,
			Parent = chip,
		})
		Paint(chipLabel, "TextColor3", "SubText")
		-- settings cog to the LEFT of the chip: opens the PRESS/HOLD mode menu
		local gear = Create("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(0, -6, 0.5, 0),
			Size = UDim2.new(0, 17, 0, 17), Text = "",
			AutoButtonColor = false, BorderSizePixel = 0, ZIndex = 4, Parent = chip,
		}, {
			Create("UICorner", { CornerRadius = UDim.new(0, 4) }),
			Create("ImageLabel", {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 0),
				Size = UDim2.new(0, 11, 0, 11), BackgroundTransparency = 1,
				Image = "rbxassetid://10734950309", ScaleType = Enum.ScaleType.Fit,
			}),
		})
		Paint(gear, "BackgroundColor3", "Card")
		local gearImg = gear:FindFirstChildWhichIsA("ImageLabel")
		local gearEntry = Paint(gearImg, "ImageColor3", "SubText")
		local bind = RegisterBind(tostring(opts.Name), opts.Keybind, function()
			state = not state
			apply(true, true)
		end, function() return state end)
		-- HOLD mode: held key = on, released = off
		bind.OnHoldStart = function()
			if not state then state = true apply(true, true) end
		end
		bind.OnHoldEnd = function()
			if state then state = false apply(true, true) end
		end
		bind.Refresh = function()
			gearEntry.r = bind.Mode == "Hold" and "Accent" or "SubText"
			Recolor(gearEntry, true)
		end
		bind.Update = function()
			chipLabel.Text = bind.Key and bind.Key.Name or "None"
			RefreshBindPanel()
		end
		bind.Update()
		bind.Refresh()
		gear.MouseButton1Click:Connect(function()
			ShowBindModeMenu(gear, bind)
		end)
		chip.MouseButton1Click:Connect(function()
			capturing = bind
			chipLabel.Text = "..."
		end)
		-- persist the mode next to the toggle's flag so configs remember it
		if opts.Flag then
			bind.ModeFlag = opts.Flag .. ".mode"
			RegisterFlag(bind.ModeFlag, "Press", function(v) SetBindMode(bind, v) end, "BindMode")
		end
	end

	RegisterFlag(opts.Flag, state, function(v) Toggle:Set(v, true) end, "Toggle")
	apply(false, false)
	if opts.Premium and not Crystalized.IsPremium then
		Premiumize(row, nameLbl, nameEntry, tostring(opts.Name))
	end
	return Toggle
end

--> SLIDER (drag + click-to-type value box) ----------------------
local function BuildSlider(section, opts)
	local min, max = opts.Min or 0, opts.Max or 100
	local decimals = opts.Decimals or 0
	local suffix = opts.Suffix or ""
	local value = math.clamp(opts.Default or min, min, max)
	local Slider = {}

	local row = MakeRow(section._holder, 52, opts.Name)
	local nameLbl, nameEntry = MakeNameLabel(row, opts.Name)
	nameLbl.Size = UDim2.new(1, -150, 0, 30)
	AttachTooltip(row, opts.Description)

	-- editable value box (click it and type!)
	local box = Create("TextBox", {
		AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -12, 0, 7),
		Size = UDim2.new(0, 64, 0, 20), BorderSizePixel = 0, TextXAlignment = Enum.TextXAlignment.Center,
		Font = Enum.Font.GothamSemibold, TextSize = 11, ClearTextOnFocus = false, Parent = row,
	}, { Create("UICorner", { CornerRadius = UDim.new(0, 5) }) })
	Paint(box, "BackgroundColor3", "Card")
	Paint(box, "TextColor3", "Text")

	-- per-element styling: Color = "Emerald"/#hex/Color3 pins the fill; Danger = true goes red + !
	local elemColor = opts.Danger and DANGER_COLOR or ResolveColor(opts.Color)
	if opts.Danger then
		nameLbl.Text = "^  " .. nameLbl.Text
		nameEntry.c = DANGER_COLOR
	end

	-- track + fill + knob
	local track = Create("Frame", {
		Position = UDim2.new(0, 12, 0, 36), Size = UDim2.new(1, -24, 0, 6),
		BorderSizePixel = 0, Parent = row,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	Paint(track, "BackgroundColor3", "ToggleOff")
	local fill = Create("Frame", {
		Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0, Parent = track,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	local fillEntry = Paint(fill, "BackgroundColor3", "Accent")
	if elemColor then fillEntry.c = elemColor Recolor(fillEntry, true) end
	local knob = Create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0, 0, 0.5, 0),
		Size = UDim2.new(0, 12, 0, 12), BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0, Parent = track,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
	-- big invisible grabber (easier on touch)
	local grab = Create("TextButton", {
		Size = UDim2.new(1, 0, 0, 24), Position = UDim2.new(0, 0, 0.5, -12),
		BackgroundTransparency = 1, Text = "", Parent = track,
	})

	local fmt = "%." .. decimals .. "f"
	local function display() box.Text = string.format(fmt, value) .. suffix end

	local function apply(animate, fireCallback)
		local scale = (max == min) and 0 or (value - min) / (max - min)
		if animate then
			Tween(fill, 0.25, { Size = UDim2.new(scale, 0, 1, 0) }, "Snappy")
			Tween(knob, 0.25, { Position = UDim2.new(scale, 0, 0.5, 0) }, "Snappy")
		else
			fill.Size = UDim2.new(scale, 0, 1, 0)
			knob.Position = UDim2.new(scale, 0, 0.5, 0)
		end
		display()
		if opts.Flag then Crystalized.Flags[opts.Flag] = value end
		if fireCallback then Fire(opts.Callback, value) end
	end

	local function setFromX(xPos)
		local scale = math.clamp((xPos - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		value = Round(min + (max - min) * scale, decimals)
		apply(false, true)
	end

	local dragging = false
	grab.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)

	box.FocusLost:Connect(function(enter)
		if enter then
			local typed = tonumber((box.Text:gsub("[^%d%.%-]", "")))
			if typed then
				value = math.clamp(Round(typed, decimals), min, max)
				apply(true, true)
			else
				display()
			end
		end
	end)

	function Slider:Set(v, noCallback)
		value = math.clamp(Round(v, decimals), min, max)
		apply(true, not noCallback)
	end
	function Slider:Get() return value end

	RegisterFlag(opts.Flag, value, function(v) Slider:Set(v, true) end, "Slider")
	apply(false, false)
	if opts.Premium and not Crystalized.IsPremium then
		Premiumize(row, nameLbl, nameEntry, tostring(opts.Name))
	end
	return Slider
end

--> DROPDOWN (single or multi select) ----------------------------
-- Multi = true   -> Default = {"ESP", "Chams"}  Callback gets a TABLE
-- Multi = false  -> Default = "ESP"             Callback gets a string
local function BuildDropdown(section, opts)
	local options = opts.Options or {}
	local multi = opts.Multi == true
	-- selected state: single -> one value, multi -> { [option] = true }
	local selected = multi and {} or opts.Default
	if multi and type(opts.Default) == "table" then
		for _, v in ipairs(opts.Default) do selected[v] = true end
	end
	local Dropdown = {}
	local open = false

	local row = MakeRow(section._holder, 38, opts.Name)
	row.ClipsDescendants = true
	AttachTooltip(row, opts.Description)

	local header = Create("TextButton", {
		Size = UDim2.new(1, 0, 0, 38), BackgroundTransparency = 1, Text = "",
		AutoButtonColor = false, ZIndex = 2, Parent = row,
	})
	local nameLbl, nameEntry = MakeNameLabel(header, opts.Name)
	nameLbl.Size = UDim2.new(0.5, -12, 1, 0)
	local valueLabel = Create("TextLabel", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -30, 0.5, 0),
		Size = UDim2.new(0.45, 0, 1, 0), BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium, TextSize = 11, TextTruncate = Enum.TextTruncate.AtEnd,
		TextXAlignment = Enum.TextXAlignment.Right, ZIndex = 2, Parent = row,
	})
	Paint(valueLabel, "TextColor3", "SubText")
	local chevron = Create("TextLabel", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -10, 0.5, 0),
		Size = UDim2.new(0, 12, 0, 12), BackgroundTransparency = 1, Rotation = 90,
		Font = Enum.Font.GothamBold, TextSize = 12, Text = ">", ZIndex = 2, Parent = row,
	})
	Paint(chevron, "TextColor3", "SubText")

	local list = Create("Frame", {
		Position = UDim2.new(0, 6, 0, 40), Size = UDim2.new(1, -12, 0, 0),
		BackgroundTransparency = 1, Parent = row,
	}, {
		Create("UIListLayout", { Padding = UDim.new(0, 3), SortOrder = Enum.SortOrder.LayoutOrder }),
		Create("UIPadding", { PaddingBottom = UDim.new(0, 6) }),
	})

	local optionRefs = {}   -- { {option=..., btn=..., label=..., check=...} }

	local function selectionArray()
		local arr = {}
		if multi then
			for _, o in ipairs(options) do if selected[o] then table.insert(arr, o) end end
		elseif selected ~= nil then
			table.insert(arr, selected)
		end
		return arr
	end

	local function selectedText()
		local arr = selectionArray()
		return #arr > 0 and table.concat(arr, ", ") or "None"
	end

	local function currentSelection()
		if multi then return selectionArray() end
		return selected
	end

	local function refreshVisuals()
		for _, ref in ipairs(optionRefs) do
			local isSel = (multi and selected[ref.option] == true) or (not multi and selected == ref.option)
			ref.label.Font = isSel and Enum.Font.GothamSemibold or Enum.Font.GothamMedium
			if ref.check then ref.check.Text = isSel and "OK" or "" end
		end
	end

	local function buildOptions()
		for _, ref in ipairs(optionRefs) do ref.btn:Destroy() end
		table.clear(optionRefs)
		for order, option in ipairs(options) do
			local optRow = Create("TextButton", {
				Size = UDim2.new(1, 0, 0, 26), Text = "", AutoButtonColor = false,
				BorderSizePixel = 0, LayoutOrder = order, Parent = list,
			}, { Create("UICorner", { CornerRadius = UDim.new(0, 5) }) })
			Paint(optRow, "BackgroundColor3", "Card")
			local check
			if multi then
				check = Create("TextLabel", {
					Position = UDim2.new(0, 8, 0, 0), Size = UDim2.new(0, 14, 1, 0),
					BackgroundTransparency = 1, Font = Enum.Font.GothamBold, TextSize = 11,
					TextXAlignment = Enum.TextXAlignment.Left, Parent = optRow,
				})
				Paint(check, "TextColor3", "Accent")
			end
			local label = Create("TextLabel", {
				Position = UDim2.new(0, multi and 26 or 10, 0, 0), Size = UDim2.new(1, -30, 1, 0),

		buildOptions()
		valueLabel.Text = selectedText()
	end

		Premiumize(row, nameLbl, nameEntry, tostring(opts.Name))
	end

--> COLOR PICKER (ADVANCED: SV pad + rainbow hue + hex + presets) --
local function BuildColorPicker(section, opts)
	local row = MakeRow(section._holder, 38, opts.Name)
	row.ClipsDescendants = true
	MakeNameLabel(row, opts.Name)
	AttachTooltip(row, opts.Description)

	local value = opts.Default or Color3.fromRGB(238, 71, 90)
	local h, s, v = Color3.toHSV(value)
	local open = false

	-- hex helpers
	local function toHex(c)
		return string.format("#%02X%02X%02X",
			math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
	end
	local function fromHex(str)
		str = tostring(str):gsub("#", ""):gsub("%s", "")
		if #str ~= 6 then return nil end
		local r = tonumber(str:sub(1, 2), 16)
		local g = tonumber(str:sub(3, 4), 16)
		local b = tonumber(str:sub(5, 6), 16)
		if r and g and b then return Color3.fromRGB(r, g, b) end
		return nil
	end

	-- the little preview swatch in the header (click to open the picker)
	local swatch = Create("TextButton", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.new(0, 34, 0, 18), Text = "", AutoButtonColor = false,
		BorderSizePixel = 0, BackgroundColor3 = value, ZIndex = 2, Parent = row,
	}, { Create("UICorner", { CornerRadius = UDim.new(0, 5) }) })
	local swatchStroke = Create("UIStroke", { Thickness = 1, Transparency = 0.4, Parent = swatch })
	Paint(swatchStroke, "Color", "Stroke")

	-- picker area (hidden until the swatch is pressed)
	local picker = Create("Frame", {
		Position = UDim2.new(0, 8, 0, 42), Size = UDim2.new(1, -16, 0, 176),
		BackgroundTransparency = 1, Parent = row,
	})
	-- the big saturation/value pad -- PRESS ANYWHERE to pick
	local sv = Create("TextButton", {
		Size = UDim2.new(1, -26, 0, 118), Text = "", AutoButtonColor = false,
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = picker,
	}, { Create("UICorner", { CornerRadius = UDim.new(0, 6) }) })
	local svWhite = Create("Frame", {
		Size = UDim2.new(1, 0, 1, 0), BorderSizePixel = 0,
		BackgroundColor3 = Color3.new(1, 1, 1), Parent = sv,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(0, 6) }),
		Create("UIGradient", {
			Color = ColorSequence.new({
				ColorSequenceKeypoint.new(0, Color3.new(1, 1, 1)),
				ColorSequenceKeypoint.new(1, Color3.fromHSV(h, 1, 1)),
			}),
		}),
	})
	Create("Frame", {
		Size = UDim2.new(1, 0, 1, 0), BorderSizePixel = 0,
		BackgroundColor3 = Color3.new(0, 0, 0), Parent = sv,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(0, 6) }),
		Create("UIGradient", {
			Rotation = 90,
			Transparency = NumberSequence.new({
				NumberSequenceKeypoint.new(0, 1),
				NumberSequenceKeypoint.new(1, 0),
			}),
		}),
	})
	local svCursor = Create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.new(0, 10, 0, 10),
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = sv, ZIndex = 5,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(1, 0) }),
		Create("UIStroke", { Thickness = 2, Color = Color3.new(0, 0, 0), Transparency = 0.6 }),
	})
	-- the rainbow bar -- PRESS to jump the hue
	local hue = Create("TextButton", {
		AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, 0, 0, 0),
		Size = UDim2.new(0, 14, 0, 118), Text = "", AutoButtonColor = false,
		BorderSizePixel = 0, BackgroundColor3 = Color3.new(1, 1, 1), Parent = picker,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(1, 0) }),
		Create("UIGradient", {
			Rotation = 90,
			Color = ColorSequence.new({
				ColorSequenceKeypoint.new(0, Color3.fromHSV(0, 1, 1)),
				ColorSequenceKeypoint.new(0.17, Color3.fromHSV(0.17, 1, 1)),
				ColorSequenceKeypoint.new(0.33, Color3.fromHSV(0.33, 1, 1)),
				ColorSequenceKeypoint.new(0.5, Color3.fromHSV(0.5, 1, 1)),
				ColorSequenceKeypoint.new(0.67, Color3.fromHSV(0.67, 1, 1)),
				ColorSequenceKeypoint.new(0.83, Color3.fromHSV(0.83, 1, 1)),
				ColorSequenceKeypoint.new(1, Color3.fromHSV(1, 1, 1)),
			}),
		}),
	})
	local hueCursor = Create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.new(1, 4, 0, 4),
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = hue, ZIndex = 5,
	}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })

	-- bottom row 1: current color square + hex input
	local preview = Create("Frame", {
		Position = UDim2.new(0, 0, 0, 124), Size = UDim2.new(0, 22, 0, 22),
		BackgroundColor3 = value, BorderSizePixel = 0, Parent = picker,
	}, { Create("UICorner", { CornerRadius = UDim.new(0, 5) }) })
	local previewStroke = Create("UIStroke", { Thickness = 1, Transparency = 0.4, Parent = preview })
	Paint(previewStroke, "Color", "Stroke")
	local hexBox = Create("TextBox", {
		Position = UDim2.new(0, 28, 0, 124), Size = UDim2.new(1, -28, 0, 22),
		BorderSizePixel = 0, Font = Enum.Font.GothamSemibold, TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left, Text = toHex(value),
		ClearTextOnFocus = false, Parent = picker,
	}, {
		Create("UICorner", { CornerRadius = UDim.new(0, 5) }),
		Create("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8) }),
	})
	Paint(hexBox, "BackgroundColor3", "Card")
	Paint(hexBox, "TextColor3", "Text")

	-- bottom row 2: quick preset swatches (the 8 accent colors)
	local updateColor -- forward-declared: preset-dot handlers below call it
	local presetNames = { "Crimson", "Violet", "Azure", "Emerald", "Amber", "Rose", "Ice", "Lime" }
	for i, name in ipairs(presetNames) do
		local dot = Create("TextButton", {
			Position = UDim2.new(0, (i - 1) * 19 + 1, 0, 154), Size = UDim2.new(0, 14, 0, 14),
			BackgroundColor3 = Crystalized.Accents[name], Text = "", AutoButtonColor = false,
			BorderSizePixel = 0, Parent = picker,
		}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
		dot.MouseButton1Click:Connect(function()
			value = Crystalized.Accents[name]
			h, s, v = Color3.toHSV(value)
			updateColor(true)
		end)
	end

	function updateColor(fireCallback)
		value = Color3.fromHSV(h, s, v)
		swatch.BackgroundColor3 = value
		preview.BackgroundColor3 = value
		svWhite.UIGradient.Color = ColorSequence.new({
			ColorSequenceKeypoint.new(0, Color3.new(1, 1, 1)),
			ColorSequenceKeypoint.new(1, Color3.fromHSV(h, 1, 1)),
		})
		if not hexBox:IsFocused() then hexBox.Text = toHex(value) end
		svCursor.Position = UDim2.new(s, 0, 1 - v, 0)
		hueCursor.Position = UDim2.new(0.5, 0, h, 0)
		if opts.Flag then Crystalized.Flags[opts.Flag] = value end
		if fireCallback then Fire(opts.Callback, value) end
	end

	-- drag handling for both pads
	local function watchDrag(pad, onMove)
		local draggingPad = false
		pad.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
				draggingPad = true
				onMove(input.Position)
			end
		end)
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
				draggingPad = false
			end
		end)
		UserInputService.InputChanged:Connect(function(input)
			if draggingPad and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
				onMove(input.Position)
			end
		end)
	end

	watchDrag(sv, function(pos)
		s = math.clamp((pos.X - sv.AbsolutePosition.X) / sv.AbsoluteSize.X, 0, 1)
		v = 1 - math.clamp((pos.Y - sv.AbsolutePosition.Y) / sv.AbsoluteSize.Y, 0, 1)
		updateColor(true)
	end)
	watchDrag(hue, function(pos)
		h = math.clamp((pos.Y - hue.AbsolutePosition.Y) / hue.AbsoluteSize.Y, 0, 0.999)
		updateColor(true)
	end)

	-- type a hex code directly
	hexBox.FocusLost:Connect(function(enter)
		if enter then
			local parsed = fromHex(hexBox.Text)
			if parsed then
				value = parsed
				h, s, v = Color3.toHSV(parsed)
				updateColor(true)
			else
				hexBox.Text = toHex(value)
			end
		end
	end)

	swatch.MouseButton1Click:Connect(function()
		open = not open
		Tween(row, 0.35, { Size = UDim2.new(1, 0, 0, open and 226 or 38) })
	end)

	local Picker = {}
	function Picker:Set(color)
		value = color
		h, s, v = Color3.toHSV(color)
		updateColor(false)
	end
	function Picker:Get() return value end
	RegisterFlag(opts.Flag, value, function(c) Picker:Set(c) end, "ColorPicker")
	updateColor(false)
	return Picker
end

local function BuildParagraph(section, opts)
	local frame = Create("Frame", {
		Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
		BackgroundTransparency = 1, Parent = section._holder,

	local startScale
	if type(options.UIScale) == "number" then
		startScale = math.clamp(options.UIScale, 0.5, 1.25)
	elseif HAS_KEYBOARD then
		startScale = 1.0 -- PC default 100%
	else
		startScale = 0.75 -- Mobile default 75%
	end

	-- apply accent/theme choices up front
	if startAccent then self:SetAccent(startAccent) end
	if options.Theme then self:SetTheme(options.Theme) end

	-- saved premium key still valid? stay unlocked this session
	if not self.IsPremium then
		local savedKey
		if type(readfile) == "function" and type(isfile) == "function" then
			if FileSafe(isfile, CONFIG_FOLDER .. "/premium.txt") then
				savedKey = FileSafe(readfile, CONFIG_FOLDER .. "/premium.txt")
			end
		else
			savedKey = memoryConfigs.__premium
		end
		if savedKey and self:IsKeyValid(savedKey) then self.IsPremium = true end

	local function spawnParticle()
		if #fxAlive > 70 then return end
		local isSparkle = fxMode == "Sparkles"
		local size = isSparkle and math.random(2, 4) or math.random(3, 6)
		local p = Create("Frame", {
			Size = UDim2.new(0, size, 0, size),
			Position = UDim2.new(math.random(), 0, isSparkle and 1.04 or -0.04, 0),
			BackgroundColor3 = isSparkle and Theme.Accent or Color3.new(1, 1, 1),
			BackgroundTransparency = 1, BorderSizePixel = 0, ZIndex = 0, Parent = fxLayer,
		}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
		table.insert(fxAlive, p)
		local dur = isSparkle and math.random(3, 6) or math.random(5, 9)
		local drift = math.random(-40, 40) / 100
		local targetY = isSparkle and -0.05 or 1.05
		-- fade in -> drift across the UI -> fade out
		TweenService:Create(p, TweenInfo.new(0.5), { BackgroundTransparency = isSparkle and 0.25 or 0.15 }):Play()
		TweenService:Create(p, TweenInfo.new(dur, Enum.EasingStyle.Linear),
			{ Position = UDim2.new(math.clamp(p.Position.X.Scale + drift, 0, 1), 0, targetY, 0) }):Play()
		if isSparkle then
			TweenService:Create(p, TweenInfo.new(dur / 2, Enum.EasingStyle.Linear), { Rotation = 180 }):Play()
		end
		task.delay(dur - 0.5, function()
			TweenService:Create(p, TweenInfo.new(0.5), { BackgroundTransparency = 1 }):Play()
		end)
		task.delay(dur, function()
			for i, f in ipairs(fxAlive) do
				if f == p then table.remove(fxAlive, i) break end
			end
			p:Destroy()
		end)
	end

	-------------------- Search --------------------
	-------------------- Search (plain + fuzzy) --------------------
	-- fuzzy: characters of the query must appear, in order ("kl aru" -> "kill aura")
	local function fuzzyMatch(text, q)
		local ti = 1
		for qi = 1, #q do
			local c = string.sub(q, qi, qi)
			if c ~= " " then
				local found = string.find(text, c, ti, true)
				if not found then return false end
				ti = found + 1
			end
		end
		return true
	end

	local function runSearch()
		local query = string.lower(searchBox.Text)
		local firstHitTab = nil
		local count = 0
		for _, tab in ipairs(Window._tabs) do
			for _, col in ipairs({ tab._left, tab._right }) do
				for _, sectionFrame in ipairs(col:GetChildren()) do
					if sectionFrame:IsA("Frame") or sectionFrame:IsA("CanvasGroup") then
						local sectionName = sectionFrame:GetAttribute("SectionName") or ""
						local anyVisible = false
						for _, row in ipairs(sectionFrame:GetChildren()) do
							local searchText = row:GetAttribute("SearchText")
							if searchText then
								local show = (query == "")
									or string.find(searchText, query, 1, true) ~= nil
									or string.find(sectionName, query, 1, true) ~= nil
									or fuzzyMatch(searchText .. " " .. sectionName, query)
								row.Visible = show
								if show then
									anyVisible = true
									count = count + 1
									if query ~= "" and not firstHitTab then firstHitTab = tab end
								end
							end
						end
						sectionFrame.Visible = (query == "") or anyVisible
							or string.find(sectionName, query, 1, true) ~= nil
					end
				end
			end
		end
		-- live result count right in the placeholder (per answer 22)
		if query == "" then
			searchBox.PlaceholderText = "Search settings..."
		else
			searchBox.PlaceholderText = count .. " match" .. (count == 1 and "" or "es")
				.. "  -  Enter to jump"
		end
		Window._searchFirstHit = firstHitTab
	end
	searchBox:GetPropertyChangedSignal("Text"):Connect(runSearch)

	-- flash-highlight the matched rows when you switch tabs mid-search
	local function flashMatches(tab)
		local query = string.lower(searchBox.Text)
		if query == "" then return end
		task.delay(0.35, function()
			for _, col in ipairs({ tab._left, tab._right }) do
				for _, sectionFrame in ipairs(col:GetChildren()) do
					if (sectionFrame:IsA("Frame") or sectionFrame:IsA("CanvasGroup")) and sectionFrame.Visible then
						for _, row in ipairs(sectionFrame:GetChildren()) do
							local searchText = row:IsA("GuiObject") and row:GetAttribute("SearchText")
							if searchText and row.Visible
								and (string.find(searchText, query, 1, true) or fuzzyMatch(searchText, query)) then
								local flash = row:FindFirstChild("SearchFlash") or Create("Frame", {
									Name = "SearchFlash", Size = UDim2.new(1, 0, 1, 0),
									BackgroundColor3 = Theme.Accent, BackgroundTransparency = 1,
									BorderSizePixel = 0, ZIndex = 1, Parent = row,
								}, { Create("UICorner", { CornerRadius = UDim.new(0, 6) }) })
								flash.BackgroundColor3 = Theme.Accent
								flash.BackgroundTransparency = 0.82
								TweenService:Create(flash, TweenInfo.new(0.7, Enum.EasingStyle.Quint), {
									BackgroundTransparency = 1,
								}):Play()
							end
						end
					end
				end
			end
		end)
	end
	Window._flashMatches = flashMatches

	-- Enter inside search jumps straight to the tab holding the first match
	-- and pulse-highlights every matched row (per answer 22)
	searchBox.FocusLost:Connect(function(enter)
		if enter and searchBox.Text ~= "" and Window._searchFirstHit then
			local tab = Window._searchFirstHit
			if tab._btn then
				for _, t in ipairs(Window._tabs) do t._setActive(false) end
				tab._setActive(true)
				tab._page.CanvasPosition = Vector2.new(0, 0)
				if tab._cascade then tab._cascade() end
				flashMatches(tab)
			end
		end
	end)

	-------------------- Auto Settings tab --------------------
	task.defer(function()
		if options.ShowSettings ~= false and Window and Window._tabs then
			BuildSettingsTab(Window)
		end
	end)

	-------------------- FPS / ping updater --------------------
	local frames, elapsed = 0, 0
	RunService.RenderStepped:Connect(function(dt)
		frames = frames + 1
		elapsed = elapsed + dt
		if elapsed >= 1 then
			local fps = math.floor(frames / elapsed + 0.5)
			local ping = 0
			pcall(function() ping = math.floor(LocalPlayer:GetNetworkPing() * 1000 + 0.5) end)
			local txt = fps .. " FPS  -  " .. ping .. " ms"
			wmStats.Text = txt
			frames, elapsed = 0, 0
		end
	end)

	-------------------- Splash screen --------------------
	local function playSplash(done)
		local splash = Create("Frame", {
			Size = UDim2.new(1, 0, 1, 0), BackgroundColor3 = Color3.fromRGB(7, 7, 10),
			BorderSizePixel = 0, ZIndex = 100, Parent = Gui,
		})
		local title = Create("TextLabel", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, -56),
			Size = UDim2.new(0, 400, 0, 40), BackgroundTransparency = 1, ZIndex = 101,
			Font = Enum.Font.GothamBold, TextSize = 30, Text = windowName,
			TextColor3 = Color3.new(1, 1, 1), TextTransparency = 1, Parent = splash,
		})
		local splashTag = Create("TextLabel", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, -28),
			Size = UDim2.new(0, 400, 0, 16), BackgroundTransparency = 1, ZIndex = 101,
			Font = Enum.Font.GothamMedium, TextSize = 12, Text = tagline,
			TextColor3 = Color3.fromRGB(150, 150, 160), TextTransparency = 1, Parent = splash,
		})
		local barBg = Create("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 8),
			Size = UDim2.new(0, 260, 0, 5), BackgroundColor3 = Color3.fromRGB(26, 26, 34),
			BorderSizePixel = 0, ZIndex = 101, Parent = splash,
		}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
		local fill = Create("Frame", {
			Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0, ZIndex = 102, Parent = barBg,
		}, { Create("UICorner", { CornerRadius = UDim.new(1, 0) }) })
		fill.BackgroundColor3 = Theme.Accent
		local stageLabel = Create("TextLabel", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 30),
			Size = UDim2.new(0, 400, 0, 14), BackgroundTransparency = 1, ZIndex = 101,
			Font = Enum.Font.GothamMedium, TextSize = 11, Text = "Loading assets...",
			TextColor3 = Color3.fromRGB(130, 130, 140), Parent = splash,
		})
		Tween(title, 0.5, { TextTransparency = 0 })
		Tween(splashTag, 0.5, { TextTransparency = 0 })
		TweenService:Create(fill, TweenInfo.new(2.2, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut),
			{ Size = UDim2.new(1, 0, 1, 0) }):Play()
		task.spawn(function()
			local stages = { "Loading assets...", "Building interface...", "Applying theme...", "Finalizing..." }
			for _, s in ipairs(stages) do
				stageLabel.Text = s
				task.wait(0.55)
			end
			Fade(splash, true, 0.4)
			task.wait(0.45)
			splash:Destroy()
			done()
		end)
	end

	-------------------- Go! --------------------
	-- splash is opt-in: boot straight into the hub unless Splash = true
	if options.Splash == true then
		playSplash(function() Window:SetVisible(true) end)
	else
		Window:SetVisible(true)
	end

	-- optional "What's new" changelog card - shows ONCE per version you bump.
	-- CreateWindow({ Changelog = { Version = "1.1.0", Lines = {"added x", "fixed y"} } })
	if options.Changelog then
		local cl = options.Changelog
		local ver = tostring(cl.Version or self.Version)
		local seenVer
		if type(readfile) == "function" and type(isfile) == "function"
		and FileSafe(isfile, CONFIG_FOLDER .. "/seenversion.txt") then
			seenVer = FileSafe(readfile, CONFIG_FOLDER .. "/seenversion.txt")
		else
			seenVer = memoryConfigs.__seenVersion
		end
		if seenVer ~= ver then
			task.delay(1.1, function()
				if not Window._main or not Window._main.Parent then return end
				Window:ShowChangelog(cl.Title, cl.Lines or {}, ver)
				FileSafe(writefile, CONFIG_FOLDER .. "/seenversion.txt", ver)
				memoryConfigs.__seenVersion = ver
			end)
		end
	end

	return Window
end

return Crystalized
